LeetCode 39&40：使用DFS匹配合适的元素

2026 · 11 · 11 · 代码编写 · 约 4 分钟

解题思路

给你一个 无重复元素 的整数数组 candidates 和一个目标整数 target ，找出 candidates 中可以使数字和为目标数 target 的 所有 不同组合 ，并以列表形式返回。你可以按 任意顺序 返回这些组合。candidates 中的 同一个 数字可以 无限制重复被选取。如果至少一个数字的被选数量不同，则两种组合是不同的。

给定一个候选人编号的集合 candidates 和一个目标数 target ，找出 candidates 中所有可以使数字和为 target 的组合。candidates 中的每个数字在每个组合中只能使用 一次 。

因为两题的描述都差不多，所以可以复用一个框架，再进行剪枝和去重。

DFS深度优先搜索

因为不限制每次选数的数量，所以不能用循环嵌套，要用dfs递归实现。具体的操作如下：

f(给定数组，目标值，开始位置，选择了的)       yes target = 0? return
                                           no f(给定数组，目标值+选的，位置+1，选择了的+现在选的)

最终代码

39题

class Solution {
public:
  vector<vector<int>> combinationSum(vector<int> &candidates, int target) {
    sort(candidates.begin(), candidates.end());
    vector<vector<int>> ans;
    vector<int> tmp;
    dfs(candidates, target, 0, tmp, ans);
    return ans;
  }

private:
  void dfs(const vector<int> &c, int target, int start,
           vector<int> &tmp, vector<vector<int>> &ans) {
    if (target == 0) {
      ans.push_back(tmp);
      return; }
    for (int i = start; i < c.size(); i++) {
      if (c[i] > target)
        break;
      tmp.push_back(c[i]);
      dfs(c, target - c[i], i, tmp, ans);
      tmp.pop_back();
    }
  }
};

40题

class Solution {
public:
  vector<vector<int>> combinationSum2(vector<int> &candidates, int target) {
    sort(candidates.begin(), candidates.end());
    vector<vector<int>> ans;
    vector<int> tmp;
    dfs(candidates, target, 0, tmp, ans);
    return ans;
  }

private:
  void dfs(const vector<int> &c, int target, int start, vector<int> &tmp, vector<vector<int>> &ans) {
    if (target == 0) {
      ans.push_back(tmp);
      return; 
    }
    for (int i = start; i < c.size(); i++) {
      if (c[i] > target)
        break;
      if (i > start && c[i] == c[i-1]) continue;
      tmp.push_back(c[i]);
      dfs(c, target - c[i], i + 1, tmp, ans);
      tmp.pop_back();
    }
  }
};

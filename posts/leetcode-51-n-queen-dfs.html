Leetcode 51：N_Queen用DFS进行布局

2026 · 11 · 12 · 代码编写 · 约 3 分钟

整体思路

按照国际象棋的规则，皇后可以攻击与之处在同一行或同一列或同一斜线上的棋子。
n 皇后问题 研究的是如何将 n 个皇后放置在 n×n 的棋盘上，并且使皇后彼此之间不能相互攻击。
给你一个整数 n ，返回所有不同的 n 皇后问题 的解决方案。

题目要求王后之间不能互相攻击，即在横方向，竖方向和斜方向上不能同时存在两个王后。
因为需要找出所有可能的情况，所以需要一步步尝试所有可能的情况，并且判定当前个字是否符合规则。
我使用深度优先搜索，每次选择一条路走到底，然后倒回上一个选择，重新选择。

使用数组筛选合适位置

因为皇后的数量和行、列数都一样，且王后可以吃掉同一直线上的棋子，所以每行/每列上有且仅有一个皇后。
因为我的dfs函数里面有参数line，每次按列枚举，所以只要保证每行没有重复的就行。我的方法是使用一个数组记录可以使用的位置，再用变量记录选择的行号，删除后调用完函数再加入。

遍历对角线排查非法放置

王后可以吃掉斜线上的棋子，所以每次根据行列选择后还要用两个循环分别遍历两条斜线，检查斜线上有没有棋子。

代码实现

#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    int n_;
    vector<vector<char>> res;
    vector<vector<vector<char>>> ans;
    vector<int> can;

    void dfs(int line, int n) {
        if(line == n) {
            ans.push_back(res);
            return;
        }
        for(int i = 0; i < (int)can.size(); i++) {
            int x = can[i];
            bool valid = true;
            
            for(int r = 0; r < line; r++) if (res[r][x] == 'Q') valid = false;
            for(int r = line - 1, c = x - 1; r >= 0 && c >= 0; r--, c--) if(res[r][c] == 'Q') valid = false;
            for(int r = line - 1, c = x + 1; r >= 0 && c < n; r--, c++) if(res[r][c] == 'Q') valid = false;

            if(!valid) continue;

            can.erase(can.begin() + i);
            res[line][x] = 'Q';
            dfs(line + 1, n);
            res[line][x] = '.';
            can.insert(can.begin() + i, x);
        }
    }

    vector<vector<string>> solveNQueens(int n) {
        n_ = n;
        res.assign(n, vector<char>(n, '.'));
        ans.clear();
        can.clear();
        for(int i = 0; i < n; i++) can.push_back(i);

        dfs(0, n);

        vector<vector<string>> result;
        for(vector<vector<char>>& board : ans) {
            vector<string> b;
            for(vector<char>& row : board) b.push_back(string(row.begin(), row.end()));
            result.push_back(b);
        }
        return result;
    }
};
